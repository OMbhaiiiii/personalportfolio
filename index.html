/**
 * MartialDisciplineEngine.java
 * Demonstrating clean OOP, immutability, and domain modeling.
 * Author: Om Sikarwar (Java Programmer & Taekwondo Black Belt)
 */
package dev.omsikarwar.core;

import java.util.List;
import java.util.Objects;

public class MartialDisciplineEngine {
    
    // Core Taekwondo Tenets modeled as Immutable State
    public enum Tenet {
        COURTESY, INTEGRITY, PERSEVERANCE, SELF_CONTROL, INDOMITABLE_SPIRIT
    }

    public record BlackBeltProfile(
        String athleteName,
        String martialRank,
        List<Tenet> coreValues
    ) {}

    public static void main(String[] args) {
        BlackBeltProfile om = new BlackBeltProfile(
            "Om Sikarwar",
            "Taekwondo Black Belt (State Competitor)",
            List.of(Tenet.values())
        );

        System.out.println("Initialized: " + om.athleteName());
        System.out.println("Rank Verified: " + om.martialRank());
        
        // Applying martial perseverance to software engineering
        om.coreValues().stream()
            .map(tenet -> "Engineering Principle > " + tenet)
            .forEach(System.out::println);
    }
}

/**
 * StoryNarrative.java
 * Software meets storytelling: Builder Pattern for Narrative Craft.
 * Author: Om Sikarwar (Personal Story Writer)
 */
package dev.omsikarwar.narrative;

public class PersonalStory {
    private final String title;
    private final String coreTheme;
    private final String takeaway;

    private PersonalStory(Builder builder) {
        this.title = builder.title;
        this.coreTheme = builder.coreTheme;
        this.takeaway = builder.takeaway;
    }

    public static class Builder {
        private String title;
        private String coreTheme;
        private String takeaway;

        public Builder title(String t) { this.title = t; return this; }
        public Builder theme(String th) { this.coreTheme = th; return this; }
        public Builder takeaway(String tk) { this.takeaway = tk; return this; }
        public PersonalStory build() { return new PersonalStory(this); }
    }
}
